ตรวจตัวอย่าง K-4 (แรงสายยึดเสา) ด้วย MATLAB/Octave

P = [0 0 12];  A = [4 -3 0];  B = [-3 4 0];
uA = (A-P)/norm(A-P);  uB = (B-P)/norm(B-P);
F = 650*uA + 390*uB          % ได้ [110 -30 -960]
R = norm(F)                  % 966.75 N
alpha_beta_gamma = acosd(F/R) % มุมทิศทาง (องศา)

วงจร RLC อนุกรม (ตัวอย่าง Z-2) ด้วย Python

import numpy as np
w = 2*np.pi*50
Z = 30 + 1j*w*0.127 + 1/(1j*w*53e-6)
I = 220/Z
for name, z in [("VR", 30), ("VL", 1j*w*0.127), ("VC", 1/(1j*w*53e-6))]:
    V = I*z
    print(name, round(abs(V), 1), "V at", round(np.degrees(np.angle(V)), 2), "deg")
S = 220*np.conj(I);  print("P =", S.real, "Q =", S.imag, "PF =", S.real/abs(S))

แขนกล (ตัวอย่าง A-4) ด้วย Python และ Robotics Toolbox

import numpy as np
L1, L2, t1, t2 = 0.5, 0.4, np.radians(30), np.radians(60)
J = np.array([[-L1*np.sin(t1)-L2*np.sin(t1+t2), -L2*np.sin(t1+t2)],
              [ L1*np.cos(t1)+L2*np.cos(t1+t2),  L2*np.cos(t1+t2)]])
tau = J.T @ np.array([0, 2*9.81])   # [8.50, 0.00] N·m
